tools/canon/canon_check.mjs
import fs from "fs";
import path from "path";
import crypto from "crypto";

const ROOT = process.cwd();
const CANON_DIR = path.join(ROOT, "canon");
const LOCK_PATH = path.join(CANON_DIR, "CANON_LOCK.json");
const ACK_PATH  = path.join(CANON_DIR, "ACKNOWLEDGEMENT.json");
const PKG_PATH  = path.join(ROOT, "package.json");

function sha256(buf) {
  return crypto.createHash("sha256").update(buf).digest("hex");
}

function readJson(p) {
  return JSON.parse(fs.readFileSync(p, "utf8"));
}

function fail(msg) {
  console.error(`\n[CANON CHECK FAILED]\n${msg}\n`);
  process.exit(1);
}

function exists(p) {
  return fs.existsSync(p);
}

function fileSha256(p) {
  return sha256(fs.readFileSync(p));
}

function main() {
  if (!exists(LOCK_PATH)) fail(`Missing canon lockfile: ${LOCK_PATH}`);
  if (!exists(ACK_PATH))  fail(`Missing acknowledgement: ${ACK_PATH}`);
  if (!exists(PKG_PATH))  fail(`Missing package.json at repo root (bad zip layout?)`);

  const lock = readJson(LOCK_PATH);
  const ack  = readJson(ACK_PATH);

  // Build ID sanity (optional but recommended)
  if (!ack.buildId || typeof ack.buildId !== "string") {
    fail(`ACKNOWLEDGEMENT.json is missing buildId (string).`);
  }

  // Validate lock integrity
  const lockRaw = fs.readFileSync(LOCK_PATH);
  const lockSha = sha256(lockRaw);
  if (ack.authorityLockSha256 !== lockSha) {
    fail(
      `ACK lock hash mismatch.\n` +
      `ACK authorityLockSha256: ${ack.authorityLockSha256}\n` +
      `Actual CANON_LOCK.json : ${lockSha}\n` +
      `Update ACKNOWLEDGEMENT.json after updating CANON_LOCK.json.`
    );
  }

  // Validate required read files are acknowledged
  const required = (lock.files || []).filter(f => f.required === true);
  const readSet = new Set((ack.readFiles || []).map(String));

  for (const f of required) {
    if (!readSet.has(f.relPath)) {
      fail(`ACK is missing required read file: ${f.relPath}`);
    }
  }

  // Validate file hashes
  for (const f of (lock.files || [])) {
    const abs = path.join(ROOT, f.relPath);
    if (!exists(abs)) fail(`Authority file missing: ${f.relPath}`);

    const actual = fileSha256(abs);
    if (actual !== f.sha256) {
      fail(
        `Authority file hash mismatch: ${f.relPath}\n` +
        `Expected: ${f.sha256}\n` +
        `Actual  : ${actual}\n` +
        `If you intentionally changed this doc, re-lock canon via: npm run canon:lock`
      );
    }
  }

  console.log(`[CANON CHECK PASSED] buildId=${ack.buildId}`);
}

main();
tools/canon/canon_lock.mjs
import fs from "fs";
import path from "path";
import crypto from "crypto";

const ROOT = process.cwd();
const CANON_DIR = path.join(ROOT, "canon");
const INDEX_PATH = path.join(CANON_DIR, "AUTHORITY_INDEX.md");
const LOCK_PATH  = path.join(CANON_DIR, "CANON_LOCK.json");

function sha256(buf) {
  return crypto.createHash("sha256").update(buf).digest("hex");
}

function fileSha256(p) {
  return sha256(fs.readFileSync(p));
}

function fail(msg) {
  console.error(`\n[CANON LOCK FAILED]\n${msg}\n`);
  process.exit(1);
}

function parseIndex(md) {
  // Format required (simple, strict):
  // - [required] canon/DECISIONS_LOG.md
  // - [required] canon/MATCH_SETTINGS_SPEC.md
  // - [optional] canon/...
  const lines = md.split(/\r?\n/);
  const out = [];
  for (const line of lines) {
    const m = line.match(/^\s*-\s*\[(required|optional)\]\s+(.+?)\s*$/i);
    if (!m) continue;
    const required = m[1].toLowerCase() === "required";
    let rel = m[2].trim();
    // Allow paths written as "canon/..." or "./canon/..."
    rel = rel.replace(/^\.\//, "");
    out.push({ required, relPath: rel });
  }
  return out;
}

function main() {
  if (!fs.existsSync(CANON_DIR)) fail(`Missing /canon directory.`);
  if (!fs.existsSync(INDEX_PATH)) fail(`Missing ${INDEX_PATH}`);

  const idxRaw = fs.readFileSync(INDEX_PATH, "utf8");
  const files = parseIndex(idxRaw);

  if (files.length === 0) {
    fail(`AUTHORITY_INDEX.md has no parsable entries. Use "- [required] canon/<file>" lines.`);
  }

  const locked = [];
  for (const f of files) {
    const abs = path.join(ROOT, f.relPath);
    if (!fs.existsSync(abs)) fail(`Missing file listed in index: ${f.relPath}`);
    locked.push({
      relPath: f.relPath,
      required: f.required,
      sha256: fileSha256(abs),
    });
  }

  const payload = {
    version: 1,
    generatedAtUtc: new Date().toISOString(),
    files: locked,
  };

  fs.writeFileSync(LOCK_PATH, JSON.stringify(payload, null, 2));
  console.log(`[CANON LOCK WRITTEN] ${LOCK_PATH}`);
}

main();
